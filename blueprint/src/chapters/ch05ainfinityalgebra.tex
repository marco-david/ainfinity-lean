\chapter{A-infinity Algebras}

An \Ainf{}-algebra is an \Ainf{}-category with a single object. This chapter is not
formalized yet. The planned definitions reuse the one-object case of the \Ainf{}-category
structures and Stasheff sums from the previous chapters.

\begin{definition}[\Ainf{}-algebra structure]
%\lean{AInfinityTheory.AInfinityAlgebraStruct}
\label{AInfinityTheory.AInfinityAlgebraStruct}
\uses{AInfinityTheory.GradedMultilinearMap, AInfinityTheory.operationTargetDeg}
Let $A$ be a $\beta$-indexed family of $R$-modules. An \Ainf{}-algebra structure on $A$
consists of, for every positive arity $n \geq 1$, a graded multilinear map
\[
  m_n : A \otimes \cdots \otimes A \to A
\]
of degree $(2-n) \cdot \mathbf{1}$. For input degrees $(d_0,\dots,d_{n-1})$, its
component lands in $A(i)$ for any degree $i$ equal to the operation target degree
$\sum_k d_k + (2-n) \cdot \mathbf{1}$.
\end{definition}

\begin{definition}[Associated \Ainf{}-category structure]
%\lean{AInfinityTheory.AInfinityAlgebraStruct.toAInfinityCategoryStruct}
\label{AInfinityTheory.AInfinityAlgebraStruct.toAInfinityCategoryStruct}
\uses{AInfinityTheory.AInfinityAlgebraStruct, AInfinityTheory.AInfinityCategoryStruct}
An \Ainf{}-algebra structure on $A$ determines a one-object \Ainf{}-category structure. Its
object type is $\mathrm{PUnit}$, its graded hom-module is the constant family $A$, and its
operations $m_n$ are those of the algebra.
\end{definition}

\begin{definition}[Stasheff sum of an \Ainf{}-algebra structure]
%\lean{AInfinityTheory.AInfinityAlgebraStruct.stasheffSum}
\label{AInfinityTheory.AInfinityAlgebraStruct.stasheffSum}
\uses{AInfinityTheory.AInfinityAlgebraStruct.toAInfinityCategoryStruct, AInfinityTheory.stasheffSum}
The arity-$n$ Stasheff sum of an \Ainf{}-algebra structure is the indexed Stasheff sum of
its associated one-object \Ainf{}-category structure, for the unique string of objects.
\end{definition}

\begin{definition}[\Ainf{}-algebra]
%\lean{AInfinityTheory.AInfinityAlgebra}
\label{AInfinityTheory.AInfinityAlgebra}
\uses{AInfinityTheory.AInfinityAlgebraStruct.stasheffSum}
An \Ainf{}-algebra is an \Ainf{}-algebra structure that satisfies the Stasheff
identities: every Stasheff sum vanishes, for every positive arity, choice of input degrees
and inputs.
\end{definition}

\begin{definition}[Associated \Ainf{}-category]
%\lean{AInfinityTheory.AInfinityAlgebra.toAInfinityCategory}
\label{AInfinityTheory.AInfinityAlgebra.toAInfinityCategory}
\uses{AInfinityTheory.AInfinityAlgebra, AInfinityTheory.AInfinityCategory}
An \Ainf{}-algebra determines a one-object \Ainf{}-category, whose underlying
\Ainf{}-category structure is the associated one-object structure.
\end{definition}
