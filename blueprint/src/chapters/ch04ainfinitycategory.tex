\chapter{A-infinity Categories}

\begin{definition}[\Ainf{}-category structure]
\lean{AInfinityTheory.AInfinityCategoryStruct}
\label{AInfinityTheory.AInfinityCategoryStruct}
\leanok
\uses{AInfinityTheory.GradedLinearQuiver, AInfinityTheory.AInfinityOperation}
An \Ainf{}-category structure over $R$ with object type $\mathrm{Obj}$ consists of a
graded $R$-linear quiver
\[
  \Hom : \mathrm{Obj} \to \mathrm{Obj} \to \grMod_R^\beta
\]
together with \Ainf{}-operations
\[
  m_n : \Hom(X_0, X_1) \otimes \dots \otimes \Hom(X_{n-1}, X_n) \to \Hom(X_0, X_n)
\]
of degree $(2-n) \cdot \mathbf{1}$ for every positive arity $n \geq 1$. No relations are
imposed on these operations.
\end{definition}

\section{Stasheff Identities}

\begin{definition}[\Ainf{}-category]
\lean{AInfinityTheory.AInfinityCategory}
\label{AInfinityTheory.AInfinityCategory}
\leanok
\uses{AInfinityTheory.AInfinityCategoryStruct, AInfinityTheory.AInfinityCategory.stasheffSum_eq_zero}
An \Ainf{}-category is an \Ainf{}-category structure whose operations satisfy the
Stasheff identities in every positive arity.
\end{definition}

\begin{proposition}[\Ainf{}-categories with vanishing higher operations]
\lean{AInfinityTheory.AInfinityCategoryStruct.ofArityLeTwo, AInfinityTheory.AInfinityCategory.ofArityLeTwo}
\label{AInfinityTheory.AInfinityCategory.ofArityLeTwo}
\leanok
\uses{AInfinityTheory.AInfinityCategory, AInfinityTheory.stasheffSum}
Let $m_1$ be a family of graded linear maps of degree $\mathbf{1}$ and $m_2$ a family of
bilinear maps of degree $0$, and set $m_n = 0$ for $n \geq 3$. Then $(m_n)$ satisfies the Stasheff identities
as soon as it does so in arities $1$, $2$ and $3$, that is, as soon as $m_1^2 = 0$, $m_1$ is a
derivation of $m_2$, and $m_2$ is associative, all with Koszul signs.
\end{proposition}

\begin{proof}
\leanok
In arity $n \geq 4$, each Stasheff term $m_{r+1+t}(\dots, m_s(\dots), \dots)$ has
$s + (r + 1 + t) = n + 1 \geq 5$, so one of its two operations has arity at least $3$ and
vanishes. In arity $3$, the four terms with an operation of arity $3$ vanish, and the remaining
two form the associativity identity.
\end{proof}

\begin{proposition}[DG categories are \Ainf{}-categories]
\lean{AInfinityTheory.DGCategory.aInfinityCategory}
\label{AInfinityTheory.DGCategory.aInfinityCategory}
\leanok
\uses{AInfinityTheory.AInfinityCategory.ofArityLeTwo}
Let $\mathcal{C}$ be a category enriched over cochain complexes of $R$-modules. For $a$ of
degree $p$ and $b$ of degree $q$, set $m_1(a) = (-1)^p\, da$ and $m_2(a, b) = (-1)^q\, a b$,
and $m_n = 0$ for $n \geq 3$. Here $ab$ is the composite of $a$ followed by $b$, in
diagrammatic order. This is an \Ainf{}-category with grading $\mathbb{Z}$.
\end{proposition}

\begin{proof}
\leanok
By the previous proposition, it suffices to check the identities in arities $1$, $2$ and $3$.
The first is $d^2 = 0$. The second follows from the Leibniz rule
$d(ab) = (da)b + (-1)^p a\, db$, which is how the differential of the tensor product of
complexes is defined. For the third, if $c$ has degree $r$, then associativity of composition
gives $m_2(m_2(a, b), c) = (-1)^r m_2(a, m_2(b, c))$, and the Koszul sign $(-1)^{r-1}$ of $c$
makes the two Stasheff terms cancel.
\end{proof}

\begin{proposition}[Linear categories are \Ainf{}-categories]
\lean{AInfinityTheory.GradedLinearQuiver.ofLinearCategory, AInfinityTheory.AInfinityCategory.ofLinearCategory}
\label{AInfinityTheory.AInfinityCategory.ofLinearCategory}
\leanok
\uses{AInfinityTheory.AInfinityCategory.ofArityLeTwo}
Let $\mathcal{C}$ be an $R$-linear category and $\beta$ a grading. Place all morphisms of
$\mathcal{C}$ in degree $0$, and set $m_2(a, b) = ab$, the composite of $a$ followed by $b$,
and $m_n = 0$ for $n \neq 2$. This is an \Ainf{}-category with grading $\beta$.
\end{proposition}

\begin{proof}
\leanok
By the proposition on vanishing higher operations, it suffices to check the identities in
arities $1$, $2$ and $3$. Since $m_1 = 0$, the first two hold trivially. For the third, if $c$
has degree $r \neq 0$ then $c = 0$; if $r = 0$, the Koszul sign $(-1)^{r-1} = -1$ of $c$ turns
the identity into $(ab)c = a(bc)$, the associativity of composition.
\end{proof}

\section{A-infinity Functors}

The source and target of an \Ainf{}-functor may be graded by different
gradings $\beta$ and $\beta'$; the functor then carries a homomorphism of
gradings $\beta \to \beta'$ along which degrees are translated.

\begin{definition}[\Ainf{}-functor data]
\lean{AInfinityTheory.AInfinityFunctorData}
\label{AInfinityTheory.AInfinityFunctorData}
\leanok
\uses{AInfinityTheory.GradedLinearQuiver, AInfinityTheory.GradingHom, AInfinityTheory.GradedMultilinearMap}
The data of an \Ainf{}-functor $F$ from a graded $R$-linear quiver graded by
$\beta$ to one graded by $\beta'$ consists of a map $F$ on objects, a homomorphism
of gradings $\beta \to \beta'$, and, for every positive arity $n$ and every
string of objects $X_0,\dots,X_n$, a graded multilinear map
\[
  \varphi_n : \Hom(X_0,X_1) \otimes \cdots \otimes \Hom(X_{n-1},X_n) \to \Hom(FX_0, FX_n)
\]
of degree $(1-n) \cdot \mathbf{1}$, where a degree $b \in \beta$ on the left
corresponds to its image in $\beta'$ on the right. No relations are imposed.
\end{definition}

\begin{definition}[\Ainf{}-functor equations]
\lean{AInfinityTheory.AInfinityFunctorData.mapCompSum, AInfinityTheory.AInfinityFunctorData.compMapSum, AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum}
\label{AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum}
\leanok
\uses{AInfinityTheory.AInfinityFunctorData, AInfinityTheory.AInfinityCategoryStruct, AInfinityTheory.signedStasheffSum, AInfinityTheory.operationTargetDeg}
Fix \Ainf{}-category structures with operations $m$ on the source and $m'$ on the target. For composable
inputs $a_1,\dots,a_n$, the left-hand side of the arity-$n$ functor equation is
the signed Stasheff sum
\[
  \sum_{r+s+t=n,\ s\geq 1} (-1)^{\|a_{r+s+1}\| + \cdots + \|a_n\|}\,
  \varphi_{r+1+t}\bigl(a_1,\dots,a_r,\, m_s(a_{r+1},\dots,a_{r+s}),\, a_{r+s+1},\dots,a_n\bigr),
\]
and the right-hand side is the sum, over all compositions $i_1 + \cdots + i_k = n$
of $n$ into positive parts, of
\[
  m'_k\bigl(\varphi_{i_1}(a_1,\dots,a_{i_1}),\dots,\varphi_{i_k}(a_{n-i_k+1},\dots,a_n)\bigr).
\]
There is no sign on the right-hand side, since in reduced degrees every
$\varphi_i$ has degree $0$. Both sides lie in $\Hom(FX_0,FX_n)$, in the image under
the grading homomorphism of the degree of $m_n(a_1,\dots,a_n)$. The functor data satisfies the
\Ainf{}-functor equations if the two sides agree in every positive arity, for
every string of objects, choice of input degrees and composable inputs.
\end{definition}

\begin{definition}[\Ainf{}-functor]
\lean{AInfinityTheory.AInfinityFunctor}
\label{AInfinityTheory.AInfinityFunctor}
\leanok
\uses{AInfinityTheory.AInfinityCategory, AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum}
An \Ainf{}-functor between \Ainf{}-categories is \Ainf{}-functor data that
satisfies the \Ainf{}-functor equations for the structure maps of the two
categories.
\end{definition}

\begin{proposition}[\Ainf{}-functors with vanishing higher components]
\lean{AInfinityTheory.AInfinityFunctorData.ofArityLeTwo, AInfinityTheory.AInfinityFunctorData.ofArityLeTwo_mapCompSum_eq_compMapSum}
\label{AInfinityTheory.AInfinityFunctorData.ofArityLeTwo_mapCompSum_eq_compMapSum}
\leanok
\uses{AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum, AInfinityTheory.AInfinityCategory.ofArityLeTwo}
Let $m$ and $m'$ be \Ainf{}-category structures on the source and the target with $m_n = 0$ and
$m'_n = 0$ for $n \geq 3$, and let $\varphi$ be \Ainf{}-functor data with $\varphi_n = 0$ for
$n \geq 3$. Then $\varphi$ satisfies the \Ainf{}-functor equations in every arity as soon as it
does so in arities $1$, $2$, $3$ and $4$, that is, as soon as
\begin{align*}
  \varphi_1(m_1(a)) &= m'_1(\varphi_1(a)), \\
  (-1)^{\|b\|} \varphi_2(m_1(a), b) + \varphi_2(a, m_1(b)) + \varphi_1(m_2(a, b))
    &= m'_2(\varphi_1(a), \varphi_1(b)) + m'_1(\varphi_2(a, b)), \\
  (-1)^{\|c\|} \varphi_2(m_2(a, b), c) + \varphi_2(a, m_2(b, c))
    &= m'_2(\varphi_2(a, b), \varphi_1(c)) + m'_2(\varphi_1(a), \varphi_2(b, c)), \\
  m'_2(\varphi_2(a, b), \varphi_2(c, e)) &= 0
\end{align*}
for all composable $a, b, c, e$.
\end{proposition}

\begin{proof}
\leanok
A term $\varphi_{r+1+t}(\dots, m_s(\dots), \dots)$ of the left-hand side in arity $n$ has
$s + (r + 1 + t) = n + 1$. It vanishes unless $s \leq 2$ and $r + 1 + t \leq 2$, so the left-hand
side vanishes for $n \geq 4$, and in arity $3$ only the two terms with $s = 2$ remain. A term
$m'_k(\varphi_{i_1}(\dots), \dots, \varphi_{i_k}(\dots))$ of the right-hand side vanishes unless
$k \leq 2$ and every $i_j \leq 2$. This leaves the compositions $(1)$ of $1$; $(2)$ and $(1, 1)$
of $2$; $(2, 1)$ and $(1, 2)$ of $3$; and $(2, 2)$ of $4$, and none for $n \geq 5$. The remaining
terms are the four identities.
\end{proof}
