\chapter{A-infinity Categories}

\begin{definition}[\Ainf{}-category structure]
\lean{AInfinityTheory.AInfinityCategoryStruct}
\label{AInfinityTheory.AInfinityCategoryStruct}
\leanok
\uses{AInfinityTheory.GradedLinearQuiver, AInfinityTheory.AInfinityOperation}
An \Ainf{}-category structure over $R$ with object type $\mathrm{Obj}$ consists of a
graded $R$-linear quiver
\[
  \Hom : \mathrm{Obj} \to \mathrm{Obj} \to \grMod_R^\beta
\]
together with \Ainf{}-operations
\[
  m_n : \Hom(X_0, X_1) \otimes \dots \otimes \Hom(X_{n-1}, X_n) \to \Hom(X_0, X_n)
\]
of degree $(2-n) \cdot \mathbf{1}$ for every positive arity $n \geq 1$. No relations are
imposed on these operations.
\end{definition}

\section{Stasheff Identities}

\begin{definition}[\Ainf{}-category]
\lean{AInfinityTheory.AInfinityCategory}
\label{AInfinityTheory.AInfinityCategory}
\leanok
\uses{AInfinityTheory.AInfinityCategoryStruct, AInfinityTheory.AInfinityCategory.stasheffSum_eq_zero}
An \Ainf{}-category is an \Ainf{}-category structure whose operations satisfy the
Stasheff identities in every positive arity.
\end{definition}

\begin{proposition}[\Ainf{}-categories with vanishing higher operations]
\lean{AInfinityTheory.AInfinityCategoryStruct.ofArityLeTwo, AInfinityTheory.AInfinityCategory.ofArityLeTwo}
\label{AInfinityTheory.AInfinityCategory.ofArityLeTwo}
\leanok
\uses{AInfinityTheory.AInfinityCategory, AInfinityTheory.stasheffSum}
Let $m_1$ be a family of graded linear maps of degree $\mathbf{1}$ and $m_2$ a family of
bilinear maps of degree $0$, and set $m_n = 0$ for $n \geq 3$. Then $(m_n)$ satisfies the Stasheff identities
as soon as it does so in arities $1$, $2$ and $3$, that is, as soon as $m_1^2 = 0$, $m_1$ is a
derivation of $m_2$, and $m_2$ is associative, all with Koszul signs.
\end{proposition}

\begin{proof}
\leanok
In arity $n \geq 4$, each Stasheff term $m_{r+1+t}(\dots, m_s(\dots), \dots)$ has
$s + (r + 1 + t) = n + 1 \geq 5$, so one of its two operations has arity at least $3$ and
vanishes. In arity $3$, the four terms with an operation of arity $3$ vanish, and the remaining
two form the associativity identity.
\end{proof}

\begin{proposition}[DG categories are \Ainf{}-categories]
\lean{AInfinityTheory.DGCategory.aInfinityCategory}
\label{AInfinityTheory.DGCategory.aInfinityCategory}
\leanok
\uses{AInfinityTheory.AInfinityCategory.ofArityLeTwo}
Let $\mathcal{C}$ be a category enriched over cochain complexes of $R$-modules. For $a$ of
degree $p$ and $b$ of degree $q$, set $m_1(a) = (-1)^p\, da$ and $m_2(a, b) = (-1)^q\, a b$,
and $m_n = 0$ for $n \geq 3$. Here $ab$ is the composite of $a$ followed by $b$, in
diagrammatic order. This is an \Ainf{}-category with grading $\mathbb{Z}$.
\end{proposition}

\begin{proof}
\leanok
By the previous proposition, it suffices to check the identities in arities $1$, $2$ and $3$.
The first is $d^2 = 0$. The second follows from the Leibniz rule
$d(ab) = (da)b + (-1)^p a\, db$, which is how the differential of the tensor product of
complexes is defined. For the third, if $c$ has degree $r$, then associativity of composition
gives $m_2(m_2(a, b), c) = (-1)^r m_2(a, m_2(b, c))$, and the Koszul sign $(-1)^{r-1}$ of $c$
makes the two Stasheff terms cancel.
\end{proof}

\begin{proposition}[Linear categories are \Ainf{}-categories]
\lean{AInfinityTheory.GradedLinearQuiver.ofLinearCategory, AInfinityTheory.AInfinityCategory.ofLinearCategory}
\label{AInfinityTheory.AInfinityCategory.ofLinearCategory}
\leanok
\uses{AInfinityTheory.AInfinityCategory.ofArityLeTwo}
Let $\mathcal{C}$ be an $R$-linear category and $\beta$ a grading. Place all morphisms of
$\mathcal{C}$ in degree $0$, and set $m_2(a, b) = ab$, the composite of $a$ followed by $b$,
and $m_n = 0$ for $n \neq 2$. This is an \Ainf{}-category with grading $\beta$.
\end{proposition}

\begin{proof}
\leanok
By the proposition on vanishing higher operations, it suffices to check the identities in
arities $1$, $2$ and $3$. Since $m_1 = 0$, the first two hold trivially. For the third, if $c$
has degree $r \neq 0$ then $c = 0$; if $r = 0$, the Koszul sign $(-1)^{r-1} = -1$ of $c$ turns
the identity into $(ab)c = a(bc)$, the associativity of composition.
\end{proof}

\section{A-infinity Functors}

The source and target of an \Ainf{}-functor may be graded by different
gradings $\beta$ and $\beta'$; the functor then carries a homomorphism of
gradings $\beta \to \beta'$ along which degrees are translated.

\begin{definition}[Composable multilinear map]
\lean{AInfinityTheory.ComposableMultilinearMap, AInfinityTheory.AInfinityCategoryStruct.toComposableMultilinearMap, AInfinityTheory.ComposableMultilinearMap.id}
\label{AInfinityTheory.ComposableMultilinearMap}
\leanok
\uses{AInfinityTheory.GradedLinearQuiver, AInfinityTheory.GradingHom, AInfinityTheory.GradedMultilinearMap}
Let $s$ be an integer. Fix graded $R$-linear quivers graded by $\beta$ and by $\beta'$, a map $f$
between their objects, and a homomorphism of gradings $\beta \to \beta'$. A composable multilinear
map $\Phi$ of degree $s$ over $f$ consists of, for every positive arity $n$ and every string of
objects $X_0,\dots,X_n$, a graded multilinear map
\[
  \Phi_n : \Hom(X_0,X_1) \otimes \cdots \otimes \Hom(X_{n-1},X_n) \to \Hom(f X_0, f X_n)
\]
of degree $(s-n) \cdot \mathbf{1}$, where a degree $b \in \beta$ on the left corresponds to its
image in $\beta'$ on the right. No relations are imposed. In reduced degrees, every $\Phi_n$ has
degree $s - 1$.

The operations $m_n$ of an \Ainf{}-category structure, of degree $(2-n) \cdot \mathbf{1}$, form a
composable multilinear map of degree $2$ over the identity. The identity $\mathbb{1}$ is the
composable multilinear map of degree $1$ over the identity whose unary component is the identity
and whose other components vanish.
\end{definition}

\begin{definition}[\Ainf{}-functor data]
\lean{AInfinityTheory.AInfinityFunctorData, AInfinityTheory.AInfinityFunctorData.toComposableMultilinearMap}
\label{AInfinityTheory.AInfinityFunctorData}
\leanok
\uses{AInfinityTheory.ComposableMultilinearMap}
The data of an \Ainf{}-functor $F$ from a graded $R$-linear quiver graded by
$\beta$ to one graded by $\beta'$ is a map $F$ on
objects, a homomorphism of gradings $\beta \to \beta'$, and a composable multilinear map of
degree $1$ over them: for every positive arity $n$ and
every string of objects $X_0,\dots,X_n$, a graded multilinear map
\[
  \varphi_n : \Hom(X_0,X_1) \otimes \cdots \otimes \Hom(X_{n-1},X_n) \to \Hom(FX_0, FX_n)
\]
of degree $(1-n) \cdot \mathbf{1}$. No relations are imposed.
\end{definition}

\begin{definition}[Composition of composable multilinear maps]
\lean{AInfinityTheory.ComposableMultilinearMap.comp, AInfinityTheory.ComposableMultilinearMap.compMarked, AInfinityTheory.AInfinityFunctorData.comp}
\label{AInfinityTheory.ComposableMultilinearMap.comp}
\leanok
\uses{AInfinityTheory.ComposableMultilinearMap, AInfinityTheory.AInfinityFunctorData}
Let $\Phi$ be a composable multilinear map of degree $s$ over $g$, and let $F$, of degree $1$,
and $D$, of degree $2$, be composable multilinear maps over the same map $f$, such that the target
of $f$ is the source of $g$. The composite $\Phi \circ F$ is the composable
multilinear map of degree $s$ over $g \circ f$ with components
\[
  (\Phi \circ F)_n(a_1,\dots,a_n) = \sum_{i_1 + \cdots + i_k = n}
  \Phi_k\bigl(F_{i_1}(a_1,\dots,a_{i_1}),\dots,F_{i_k}(a_{n-i_k+1},\dots,a_n)\bigr),
\]
summed over all compositions of $n$ into positive parts. The composite $\Phi \circ (F \mid D)$ is
the composable multilinear map of degree $s + 1$ over $g \circ f$ with components
\[
  (\Phi \circ (F \mid D))_n(a_1,\dots,a_n) = \sum_{i_1 + \cdots + i_k = n}\ \sum_{j = 1}^{k}
  (-1)^{\|a_{i_1 + \cdots + i_j + 1}\| + \cdots + \|a_n\|}\,
  \Phi_k\bigl(F_{i_1}(\dots),\dots,D_{i_j}(\dots),\dots,F_{i_k}(\dots)\bigr),
\]
in which the $j$-th input of $\Phi_k$ is the value of $D$, instead of $F$, on the $j$-th block.
The composite $\Phi \circ F$ is linear in $\Phi$ and polynomial in $F$, and
$\Phi \circ (F \mid D)$ is its derivative at $F$ in the direction $D$. The composite $G \circ F$
of \Ainf{}-functor data has the composite of the components of $G$ and of $F$ as its components.
\end{definition}

\begin{definition}[\Ainf{}-functor equations]
\lean{AInfinityTheory.AInfinityFunctorData.mapCompSum, AInfinityTheory.AInfinityFunctorData.compMapSum, AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum, AInfinityTheory.AInfinityFunctorData.compMapSum_eq_comp_map_v, AInfinityTheory.AInfinityFunctorData.mapCompSum_eq_compMarked_map_v, AInfinityTheory.AInfinityFunctor.compMarked_id_eq_comp}
\label{AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum}
\leanok
\uses{AInfinityTheory.AInfinityFunctorData, AInfinityTheory.AInfinityCategoryStruct, AInfinityTheory.signedStasheffSum, AInfinityTheory.targetDeg, AInfinityTheory.ComposableMultilinearMap.comp}
Fix \Ainf{}-category structures with operations $m$ on the source and $m'$ on the target. For composable
inputs $a_1,\dots,a_n$, the left-hand side of the arity-$n$ functor equation is
the signed Stasheff sum
\[
  \sum_{r+s+t=n,\ s\geq 1} (-1)^{\|a_{r+s+1}\| + \cdots + \|a_n\|}\,
  \varphi_{r+1+t}\bigl(a_1,\dots,a_r,\, m_s(a_{r+1},\dots,a_{r+s}),\, a_{r+s+1},\dots,a_n\bigr),
\]
and the right-hand side is the sum, over all compositions $i_1 + \cdots + i_k = n$
of $n$ into positive parts, of
\[
  m'_k\bigl(\varphi_{i_1}(a_1,\dots,a_{i_1}),\dots,\varphi_{i_k}(a_{n-i_k+1},\dots,a_n)\bigr).
\]
The right-hand side is the value of the composite $m' \circ F$, and the left-hand side is the
value of the composite $F \circ (\mathbb{1} \mid m)$: its terms vanish unless all blocks other
than the marked one have size one. So the functor equations read
$F \circ (\mathbb{1} \mid m) = m' \circ F$. There is no sign on the right-hand side, since in
reduced degrees every
$\varphi_i$ has degree $0$. Both sides lie in $\Hom(FX_0,FX_n)$, in the image under
the grading homomorphism of the degree of $m_n(a_1,\dots,a_n)$. The functor data satisfies the
\Ainf{}-functor equations if the two sides agree in every positive arity, for
every string of objects, choice of input degrees and composable inputs.
\end{definition}

\begin{definition}[\Ainf{}-functor]
\lean{AInfinityTheory.AInfinityFunctor}
\label{AInfinityTheory.AInfinityFunctor}
\leanok
\uses{AInfinityTheory.AInfinityCategory, AInfinityTheory.AInfinityFunctor.mapCompSum_eq_compMapSum}
An \Ainf{}-functor between \Ainf{}-categories is \Ainf{}-functor data that
satisfies the \Ainf{}-functor equations for the structure maps of the two
categories.
\end{definition}

\begin{theorem}[Associativity and unit laws of the composition]
\lean{AInfinityTheory.ComposableMultilinearMap.comp_assoc, AInfinityTheory.ComposableMultilinearMap.compMarked_comp, AInfinityTheory.ComposableMultilinearMap.comp_compMarked, AInfinityTheory.ComposableMultilinearMap.comp_id, AInfinityTheory.ComposableMultilinearMap.id_comp}
\label{AInfinityTheory.ComposableMultilinearMap.comp_assoc}
\leanok
\uses{AInfinityTheory.ComposableMultilinearMap.comp}
Let $\Phi$ be a composable multilinear map, let $G$ and $F$ have degree $1$ and $E$ and $D$ have
degree $2$, with $E$ over the same map as $G$ and $D$ over the same map as $F$, composable in
this order. Then
\begin{align*}
  (\Phi \circ G) \circ F &= \Phi \circ (G \circ F), \\
  (\Phi \circ (G \mid E)) \circ F &= \Phi \circ (G \circ F \mid E \circ F), \\
  (\Phi \circ G) \circ (F \mid D) &= \Phi \circ (G \circ F \mid G \circ (F \mid D)),
\end{align*}
and $\Phi \circ \mathbb{1} = \Phi$ and $\mathbb{1} \circ F = F$.
\end{theorem}

\begin{proof}
\leanok
Both sides of each of the three rules are sums of terms
$\Phi_k\bigl(Q(P(\dots),\dots,P(\dots)),\dots,Q(P(\dots),\dots,P(\dots))\bigr)$, in which each $Q$
is $G$ or $E$ and each $P$ is $F$ or $D$. On the left, such a
term is indexed by a composition $c$ of $n$ and a composition of the number of blocks of $c$; on
the right, by a composition of $n$ and a composition of each of its blocks. These index sets are
in bijection, as for the composition of formal multilinear series, and the terms they index
agree. The second and the third rule are the derivatives of the first in $G$ and in $F$; in them
the bijection carries the marked block along, and the Koszul signs agree. For the unit laws, only
the composition into blocks of size one, respectively the composition with a single block,
contributes.
\end{proof}

\begin{theorem}[Composition of \Ainf{}-functors]
\lean{AInfinityTheory.AInfinityFunctor.comp, AInfinityTheory.AInfinityFunctor.comp_mapCompSum_eq_compMapSum, AInfinityTheory.AInfinityFunctor.comp_assoc}
\label{AInfinityTheory.AInfinityFunctor.comp}
\leanok
\uses{AInfinityTheory.AInfinityFunctor, AInfinityTheory.ComposableMultilinearMap.comp_assoc}
The composite $G \circ F$ of \Ainf{}-functors satisfies the \Ainf{}-functor equations, so it is an
\Ainf{}-functor. The composition of \Ainf{}-functors is associative.
\end{theorem}

\begin{proof}
\leanok
With the operations $m$, $m'$ and $m''$ of the three categories,
\begin{align*}
  (G \circ F) \circ (\mathbb{1} \mid m)
  &= G \circ (F \mid F \circ (\mathbb{1} \mid m))
  = G \circ (F \mid m' \circ F) \\
  &= (G \circ (\mathbb{1} \mid m')) \circ F
  = (m'' \circ G) \circ F
  = m'' \circ (G \circ F).
\end{align*}
The first equality is the third rule and a unit law, the second the functor equations of $F$, the
third the second rule and a unit law, the fourth the functor equations of $G$, and the last the
associativity. Associativity of the composition of \Ainf{}-functors is
the case of the first rule where $\Phi$ is an \Ainf{}-functor.
\end{proof}
